# WordToMD Web App Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a zero-build static web app in `wordtomd/` that converts one `.docx` to Markdown in-browser with preview + download.

**Architecture:** Single `index.html` + `app.js` + `style.css` using `mammoth.browser` for docx→HTML and `Turndown+GFM` for HTML→Markdown, images collected via mammoth callback and zipped with JSZip, preview rendered with `marked`.

**Tech Stack:** Vanilla HTML/CSS/JS, mammoth.browser.min.js 1.9.1, turndown 7.2.0 + turndown-plugin-gfm 1.0.2, marked 12.x, jszip 3.10.1 (all via CDN with local `vendor/` fallback), pytest for smoke tests.

## Global Constraints

- No backend — pure static, all conversion in-browser.
- Single `.docx` at a time, 50MB limit, `.doc` rejected in v1.
- Full fidelity: headings h1-h6, bold/italic, lists, tables (GFM), links, images.
- Images emitted as `images/image-N.ext`, downloadable via `.zip` (md + images/).
- Zero build step; run via `python3 -m http.server`; deployable to GitHub Pages.
- YAGNI: no batch, no .doc/.pdf/.odt, no auth, no persistence.

```js
let lastResult = null;
const images = [];
function buildTurndown(){
  // eslint-disable-next-line no-undef
  const td = new TurndownService({ headingStyle: "atx", codeBlockStyle: "fenced", bulletListMarker: "-" });
  // eslint-disable-next-line no-undef
  if (typeof turndownPluginGfm !== "undefined") td.use([turndownPluginGfm.gfm]);
  td.addRule("underline", { filter: ["u"], replacement: (c) => `_${c}_` });
  return td;
}
function htmlToMarkdown(html){
  const clean = html.replace(/<o:p>.*?<\/o:p>/g, "");
  return buildTurndown().turndown(clean).replace(/\n{3,}/g, "\n\n").trim() + "\n";
}
async function convertArrayBuffer(arrayBuffer){
  images.length = 0;
  let counter = 0;
  // eslint-disable-next-line no-undef
  const out = await mammoth.convertToHtml({ arrayBuffer }, {
    styleMap: ["p[style-name='Heading 1'] => h1:fresh","p[style-name='Heading 2'] => h2:fresh","p[style-name='Heading 3'] => h3:fresh"],
    convertImage: mammoth.images.imgElement(async (img) => {
      counter += 1;
      const buf = await img.readAsArrayBuffer();
      const ext = (img.contentType.split("/")[1] || "png").split("+")[0];
      const filename = `images/image-${counter}.${ext}`;
      images.push({ filename, contentType: img.contentType, buffer: buf });
      return { src: filename };
    })
  });
  const markdown = htmlToMarkdown(out.value);
  lastResult = { markdown, html: out.value, images: [...images], messages: out.messages };
  return lastResult;
}
async function handleFile(file){
  if (!file) return null;
  if (!/\.docx$/i.test(file.name)) { setStatus("Please upload a .docx file (.doc not supported in v1)."); return null; }
  if (file.size > 50 * 1024 * 1024) { setStatus("File too large — 50MB limit."); return null; }
  setStatus(`Converting ${file.name}...`);
  try {
    const buf = await file.arrayBuffer();
    const res = await convertArrayBuffer(buf);
    document.getElementById("rawOutput").value = res.markdown;
    ["copyBtn","downloadMdBtn","downloadZipBtn","clearBtn"].forEach(id=>document.getElementById(id).disabled=false);
    document.getElementById("fileMeta").textContent = `${file.name} — ${(file.size/1024).toFixed(1)} KB, ${res.images.length} image(s)`;
    setStatus(res.markdown.trim() ? "Done." : "No convertible content found.");
    renderPreview(res.markdown);
    return res;
  } catch (e) {
    console.error(e);
    setStatus("Could not parse this file. Try re-saving it in Word as .docx.");
    return null;
  }
}
function renderPreview(md){
  const el = document.getElementById("preview");
  try {
    // eslint-disable-next-line no-undef
    el.innerHTML = (typeof marked !== "undefined") ? marked.parse(md) : `<pre>${md.replace(/</g,"&lt;")}</pre>`;
  } catch { el.textContent = md; }
}
```

```js
function currentMarkdown(){ return document.getElementById("rawOutput").value || ""; }
async function copyMarkdown(){
  const md = currentMarkdown();
  if (!md) { setStatus("Nothing to copy yet."); return false; }
  try { await navigator.clipboard.writeText(md); setStatus("Copied to clipboard."); return true; }
  catch {
    const ta = document.getElementById("rawOutput"); ta.select();
    try { document.execCommand("copy"); setStatus("Copied to clipboard."); return true; }
    catch { setStatus("Copy blocked — select the text manually."); return false; }
  }
}
function downloadMarkdown(){
  const md = currentMarkdown();
  if (!md) { setStatus("Nothing to download yet."); return; }
  const blob = new Blob([md], { type: "text/markdown" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob);
  a.download = (document.getElementById("fileMeta").textContent.split(" — ")[0] || "document.docx").replace(/\.docx$/i, "") + ".md";
  a.click(); setTimeout(() => URL.revokeObjectURL(a.href), 2000);
}
async function downloadZip(){
  const md = currentMarkdown();
  if (!md) { setStatus("Nothing to download yet."); return; }
  // eslint-disable-next-line no-undef
  if (typeof JSZip === "undefined") { downloadMarkdown(); return; }
  // eslint-disable-next-line no-undef
  const zip = new JSZip();
  const base = (document.getElementById("fileMeta").textContent.split(" — ")[0] || "document.docx").replace(/\.docx$/i, "");
  zip.file(base + ".md", md);
  for (const img of (lastResult?.images || [])) zip.file(img.filename, img.buffer);
  const blob = await zip.generateAsync({ type: "blob" });
  const a = document.createElement("a");
  a.href = URL.createObjectURL(blob); a.download = base + ".zip"; a.click();
  setTimeout(() => URL.revokeObjectURL(a.href), 2000);
}
function clearAll(){
  document.getElementById("rawOutput").value = "";
  document.getElementById("preview").innerHTML = "<em>Nothing converted yet.</em>";
  document.getElementById("fileMeta").textContent = "";
  document.getElementById("fileInput").value = "";
  ["copyBtn","downloadMdBtn","downloadZipBtn","clearBtn"].forEach(id=>document.getElementById(id).disabled=true);
  setStatus("");
}
```
